module aes_128_masked_core (
    input  wire         clk, rst, start,
    input  wire [127:0] plaintext,  // 128-bit Sensitive Plaintext
    input  wire [127:0] key,        // 128-bit Cipher Key
    input  wire [127:0] mask_in,    // Random Mask Share M
    output reg  [127:0] ciphertext, // NIST Compliant Output
    output reg          done
);
  // Two-Share State Registers (Split Datapath)
  reg [127:0] state_share1; // Masked State: S' = P ^ M
  reg [127:0] state_share2; // Mask State:   M

  // Protected Non-Linear SubBytes with Mask Refresh
  masked_sub_bytes u_msub (
      .in_share1(state_share1), .in_share2(state_share2),
      .out_share1(sbox_share1), .out_share2(sbox_share2)
  );
  // Round 10: Recombination C = state_share1 ^ state_share2
endmodule
